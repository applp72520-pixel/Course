def is_terminal(state): 
	x = state[0]
	status = False
	if x >= x_max:
		status = True
	return status

def step(state, action):
	x, v = state
	if is_terminal(state):
		x_new = x_max
		v_new = 0
		reward = 0
	else:
		reward = -1.0
		v_new = -0.0025 * np.cos(3 * x) + v + 0.001 * action 
		v_new = np.clip(v_new, v_min, v_max)
		x_new = x + v_new
		x_new = np.clip(x_new, x_min, x_max)
		if x_new == x_min:
			v_new = 0.0
		if x_new == x_max:
			v_new = 0.0
			reward = 0.0
	next_state = [x_new, v_new]
	return {"next_state": next_state, "reward": reward}

def feature_active_index(state, move, x_pts, v_pts):
	m = len(x_pts) * len(v_pts)
	all_index = list(np.arange(3 * m)) 
	if move == -1:
		index = all_index[:m]
	elif move == 0:
		index = all_index[m:(m+m)]
	else:
		index = all_index[(m+m):]
	return index

#x_sd = 0.1  # not a bad choice

x_sd = 0.1   # good choice
v_sd = 0.05  # good choice

sd = [x_sd, v_sd]

def feature_calculation(state, x_pts, v_pts, sd):
	x, v = state
	feature_values = np.zeros(shape = (len(x_pts), len(v_pts)), dtype = float)
	if is_terminal(state) is False:
		x_sd, v_sd = sd
		for i in range(len(x_pts)):
			c1 = x_pts[i]
			for j in range(len(v_pts)):
				c2 = v_pts[j]
				t1 = ((x - c1) / x_sd)**2 + ((v - c2) / v_sd)**2
				t2 = np.exp(-0.5 * t1)
				feature_values[i, j] = t2
	feature_values = feature_values.flatten()
	return feature_values

def feature_calculation_theta(state, move, theta, x_pts, v_pts, sd):
	x, v = state
	feature_values = feature_calculation(state, x_pts, v_pts, sd)
	# Find active feature indices.
	index = feature_active_index(state, move, x_pts, v_pts)
	# Find the feature vector.
	feature = np.zeros(len(theta))
	feature[index] = feature_values
	return feature

sigmoid = lambda x: 1.0 / (1.0 + np.exp(-np.clip(x, -5, 5)))

def action_prob(state, theta, x_pts, v_pts, sd):
	action = [-1, 0, 1]
	m = len(action)
	preference = np.zeros(m, dtype = float)
	for i in range(m):
		move = action[i]
		feature = feature_calculation_theta(state, move, theta, x_pts, v_pts, sd)
		preference[i] = np.sum(theta * feature)
	#print("preference = ", preference)
	preference_sig = sigmoid(preference)
	#print("preference_sig = ", preference_sig)
	preference_sig_sum = np.sum(preference_sig)
	prob = preference_sig / preference_sig_sum
	#print("prob = ", prob)
	return prob

def grad_action_prob(state, theta, x_pts, v_pts, sd): 
	action = [-1, 0, 1]
	m = len(action)
	preference = np.zeros(m, dtype = float)
	all_features = np.zeros(shape = (m, len(theta)), dtype = float)
	for i in range(m):
		move = action[i]
		feature = feature_calculation_theta(state, move, theta, x_pts, v_pts, sd)
		preference[i] = np.sum(theta * feature)
		all_features[i,:] = feature
	sig_preference = sigmoid(preference)
	sig_preference_sum = np.sum(sig_preference)
	prob = sig_preference / sig_preference_sum
	t1 = np.zeros(shape = (m, len(theta)), dtype = float)
	for i in range(m):
		t1[i] = (1.0 - sig_preference[i]) * all_features[i]
	t2 = np.zeros(len(theta), dtype = float)
	for i in range(m):
		t2 = t2 + prob[i] * t1[i]	
	all_grad = np.zeros(shape = (m, len(theta)), dtype = float)
	for i in range(m):
		#print("i = ", i)
		#print("prob[i] = ", prob[i])
		all_grad[i, :] = prob[i] * (t1[i] - t2) 
	#print("all_grad = ", all_grad)
	#print("max all_grad = ", np.max(np.abs(all_grad)))
	return all_grad

soft_temp = 0.4

def assign_prob(original_prob, soft_temp):
	tmp_prob = np.power(original_prob, 1 / soft_temp)
	new_prob = tmp_prob / np.sum(tmp_prob)
	return new_prob

			# Update the weights of the policy function.
			all_grad = grad_action_prob(state, theta, x_pts, v_pts, sd)
			move_index = move + 1
			grad = all_grad[move_index]

			theta = theta + alpha_theta * eta * delta * grad / move_prob
			eta = discount * eta

		else:

			next_feature_w = feature_calculation(next_state, x_pts, v_pts, sd)
			next_v_hat = np.sum(w * next_feature_w)
			delta = reward + discount * next_v_hat - current_v_hat
			w_old = np.copy(w)

			# Update the weights of the policy function.

			eta = discount * eta
