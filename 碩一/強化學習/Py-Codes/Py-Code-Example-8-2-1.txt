# building environment

x_min = -1.2; x_max = 0.5
v_min = -0.07; v_max = 0.07

def is_terminal(state): 
	x = state[0]
	status = False
	if x >= x_max:
		status = True
	return status

def step(state, action):
	x, v = state
	if is_terminal(state):
		x_new = x_max
		v_new = 0
		reward = 0
	else:
		reward = -1.0
		v_new = -0.0025 * np.cos(3 * x) + v + 0.001 * action 
		v_new = np.clip(v_new, v_min, v_max)
		x_new = x + v_new
		x_new = np.clip(x_new, x_min, x_max)
		if x_new == x_min:
			v_new = 0.0
		if x_new == x_max:
			v_new = 0.0
			reward = 0.0
	next_state = [x_new, v_new]
	return {"next_state": next_state, "reward": reward}

# Set the tiling parameters.
# Use the simplest horizontal shift with augmented v-tilings (v-dimensional augmented)
# so that the original region can be covered by the union of all tilings.

num_tilings = 5

num_x_partitions = 8
num_v_partitions = 6

num_tiles = num_x_partitions * num_v_partitions

x_max - x_min
v_max - v_min

v_width = v_length / num_v_partitions

# Construct the first tiling.

x_eps = 0.01
v_eps = 0.005

x_tiling_min = x_min - x_width + x_eps
v_tiling_min = v_min - v_width + v_eps

x_points = np.arange(start = x_tiling_min, stop = x_tiling_min + num_x_partitions * x_width + eps, step = x_width)
v_points = np.arange(start = v_tiling_min, stop = v_tiling_min + num_v_partitions * v_width + eps, step = v_width)

# formula: x_tiling_min + (num_tilings - 1) * x_offset + x_length = x_max + x_width - x_eps

x_offset = ((x_max + x_width - x_eps) - x_tiling_min - x_length) / (num_tilings - 1)
v_offset = 0.0

intervals = []

for j in range(len(v_points)-1):
	v_left = v_points[j]
	v_right = v_points[j+1]
	v_interval = [v_left, v_right]
	for i in range(len(x_points)-1):
		x_left = x_points[i]
		x_right = x_points[i+1]
		x_interval = [x_left, x_right]
		interval = [x_interval, v_interval]
		intervals.append(interval)

np.round(intervals, 4)
len(intervals)

# Construct the remaining tilings by adding offsets.

# Any tile beyond the specified x-bounds and v-bounds will not be included.

intervals_original = deepcopy(intervals)
np.round(intervals_original, 4)

for k in range(1, num_tilings):
	for i in range(len(intervals_original)):
		x_interval = deepcopy(intervals_original[i][0])
		v_interval = deepcopy(intervals_original[i][1])
		x_interval[0] = x_interval[0] + k * x_offset
		x_interval[1] = x_interval[1] + k * x_offset
		v_interval[0] = v_interval[0] + k * v_offset
		v_interval[1] = v_interval[1] + k * v_offset
		if (x_interval[0] <= x_max) & (v_interval[0] <= v_max):
			interval = [x_interval, v_interval]
			intervals.append(interval)
		else:

num_intervals

# features: 3 possible actions: 1/3 for action = -1; 1/3 for action = 0; 1/3 for action = 1

num_features = int(3 * num_intervals)  
num_features

status_in = lambda state, I: (I[0][0] <= state[0] < I[0][1]) & (I[1][0] <= state[1] < I[1][1])

def feature_active_index(state, move):
	if is_terminal(state):
		index = []
	else:
		if move == -1:
			start_index = 0
		elif move == 0:
			start_index = num_intervals
		else:
			start_index = int(2 * num_intervals)
		index = []
		for j in range(num_intervals):
			status = status_in(state, intervals[j])
			if status == True:
				index.append(start_index + j)
	return index

def q_value(state, move, w):
	if is_terminal(state):
		q_hat = 0.0
	else:
		features = np.zeros(len(w))
		index = feature_active_index(state, move)
		features[index] = 1.0
		q_hat = np.sum(w[index] * features[index])
	return q_hat

def greedy_move(state, w):
	if is_terminal(state):
		move = 0
	else:
		q_reverse = q_value(state, -1, w)
		q_zero = q_value(state, 0, w)
		q_forward = q_value(state, 1, w)
		q_max = q_zero
		move = 0
		if q_reverse > q_max:
			q_max = q_reverse
			move = -1
		if q_forward > q_max:
			q_max = q_forward
			move = 1
	return move

def eps_greedy_move(state, w):
	if is_terminal(state):
		move = 0
	else:
		prob_eps = 0.1
		rand_num = np.random.uniform(low = 0, high = 1, size = 1)
		if rand_num < prob_eps:
			move = int(np.random.choice([-1, 0, 1], size = 1))
		else:
			move = greedy_move(state, w)
	return move

	v = 0.0  # initial velocity
	state = [float(x), v]  # initial state
	print("initial_state = ", state)
	state_trajectory.append(state)
	# Choose an initial action using eps-greedy policy.
	move = eps_greedy_move(state, w)
	# Initialize the eligibility trace.
	z = np.zeros(len(w), dtype = float)
	status = is_terminal(state)
	while status == False:
		step.obj = step(state, move)
		next_state = step.obj['next_state']
		reward = step.obj['reward']
		if is_terminal(next_state):
			print("final_state = ", next_state)
			q_hat = q_value(state, move, w)
			index = feature_active_index(state, move)
			features = np.zeros(len(w))
			features[index] = 1.0
			z = discount * lam * z + features
			w = w + alpha * (reward - q_hat) * z
			status = True
		else: 
			# Choose the next action using eps-greedy policy.
			next_move = eps_greedy_move(next_state, w)
			current_q_hat = q_value(state, move, w)
			next_q_hat = q_value(next_state, next_move, w)
			index = feature_active_index(state, move)
			features = np.zeros(len(w))
			features[index] = 1.0
			z = discount * lam * z + features
			w = w + alpha * (reward + discount * next_q_hat - current_q_hat) * z
			state = next_state
			move = next_move
			#print("state = ", state, "  move = ", move)
			state_trajectory.append(state)
			status = is_terminal(state)
			if len(state_trajectory) == max_steps:
				status = True
	num_steps[episode] = len(state_trajectory) - 1
	print("number of steps = ", num_steps[episode])
