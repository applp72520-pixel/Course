# Import packages and functions.

import numpy as np
import matplotlib.pyplot as plt

which = lambda status: np.arange(len(status))[status]

##################

n = 5  # number of states

discount = 0.95
eps = 1.0e-5  # should be small enough
prob_eps = 0.1

# action: ["left", "right"]: [0, 1]

action_word = np.array(["left", "right"])

action = np.array([0, 1])
action_num = 2

action_prob = np.array([0.6, 0.4])

state_value = np.zeros(n)

state_action_value = np.zeros(shape = (n, action_num))

# greedy action expressed by one-hot encoding

greedy_action = np.zeros(shape = (n, action_num))

visit_num = np.zeros(shape = (n, action_num), dtype = int)
return_accum = np.zeros(shape = (n, action_num))

# environment

def is_terminal(state): 
	x = state
	status = False
	if x == 4:
		status = True
	return status

def get_initial():
	bingo = True
	while bingo:
		i = np.random.randint(low = 0, high = n)
		state = i
		if is_terminal(state): 
			continue
		else:
			bingo = False
	return state

def step(state, action):
	x = state
	if is_terminal(x):
		next_state = x
		reward = 0.0
	elif x == 0:
		next_state = x + 1
		reward = -1.0
	else:
		reward = -1.0
		if action == 0: # move left 
			next_state = x - 1
		if action == 1: # move right 
			next_state = x + 1
		if is_terminal(next_state):
			reward = 0.0
	return {"next_state": next_state, "reward": reward}

# initial policy

action_prob = np.zeros(shape = (n, action_num))
for i in range(n):
	state = i
	if is_terminal(state):
		continue
	else:
		action_prob[i, :] = 0.5

#episode_num = 100000

episode_num = 100000  # for testing purpose only

seed = 543
np.random.seed(seed)  # Set random seed for reproducibility.

for episode in range(episode_num):
	#print("")
	#print("episode = ", episode)
	# some initialization
	#state_trajectory = []
	state_action_trajectory = []
	reward_seq = []
	# Get a random initial state: Exploring starts.
	state = get_initial()
	#state_trajectory.append(state)
	# Game starts!
	status = is_terminal(state)
	while status == False:
		# Choose an action using eps-greedy policy.
		rand_num = np.random.uniform(size = 1)
		if rand_num < prob_eps:
			move = int(np.random.choice(action, size = 1))
		else:
			prob = action_prob[state, :]
			prob_max = np.max(prob)
			index_max = which(np.abs(prob - prob_max) < eps)
			if len(index_max) == 1:
				move = index_max
			else:
				move = np.random.choice(index_max, size = 1)
		state_action = [state, move]
		state_action_trajectory.append(state_action)
		step_obj = step(state, move)
		next_state = step_obj['next_state']
		reward = step_obj['reward']
		#state_trajectory.append(next_state)
		reward_seq.append(reward)
		state = next_state
		status = is_terminal(state)
	#print("state_trajectory = ", state_trajectory)
	#print("state_action_trajectory = ", state_action_trajectory)
	#print("reward_seq = ", reward_seq)
	# Update the state-action-value and state-value, and find the greedy actions.
	m = len(state_action_trajectory)
	#print("m = ", m)
	G = 0
	for i in range(m-1, -1, -1):
		#print("i = ", i)
		G = discount * G + reward_seq[i]
		#print("G = ", G)
		# Estimate the state-action-value of the policy.
		state_action = state_action_trajectory[i]
		# Update the state-action-value.
		k0, k1 = state_action
		#print("state_action = ", state_action)
		visit_num[k0, k1] = visit_num[k0, k1] + 1
		return_accum[k0, k1] = return_accum[k0, k1] + G
		state_action_value[k0, k1] = return_accum[k0, k1] / visit_num[k0, k1]
		#print("visit_num = ", visit_num[k0, k1])
		#print("return_accum = ", return_accum[k0, k1])
		#print("state_action_value = ", state_action_value[k0, k1])
		q_values = state_action_value[k0, :] 
		q_max = np.max(q_values)
		index_max = which(np.abs(q_values - q_max) < eps)
		index_max_length = len(index_max)
		action_prob[k0, :] = 0.0
		action_prob[k0, index_max] = 1.0 / index_max_length
		state_value[k0] = q_max
		greedy_action[k0, :] = np.zeros(action_num)
		greedy_action[k0, index_max] = 1

# final results

np.round(state_value, 4)

np.round(state_action_value, 4)

visit_num

visit_num = visit_num.flatten()
index = which(visit_num > 0)

plt.boxplot(visit_num[index])
plt.title('Number of state-action visits')
plt.xticks([])
plt.show()

for i in range(n): 
	if is_terminal(i):
		print("state = ", i, "  greedy action = ", [])
	else:
		index_max = which(greedy_action[i, :] == 1)
		print("state = ", i, "  greedy action = ", action_word[index_max])

# trajectory using greedy actions

i = 0
state = get_initial()
status = is_terminal(state)
while status == False:
	index_max = which(greedy_action[state, :] == 1)
	move = index_max[0]
	print("i = ", i, "  state = ", state, "  move = ", action_word[move])
	step_obj = step(state, move)
	next_state = step_obj['next_state']
	reward = step_obj['reward']
	state = next_state
	i = i + 1
	status = is_terminal(state)
	if status == True:
		print("i = ", i, "  state = ", state, "  move = ", "halt")
