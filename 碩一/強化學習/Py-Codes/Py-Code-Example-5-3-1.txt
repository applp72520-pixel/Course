# Python code for Example 5.2.1: Two-dimensional maze

# Find the state values and state-actions values using n-step bootstrapping.

##################

# Import packages and functions.

import numpy as np

which = lambda status: np.arange(len(status))[status]

##################

nrow = 5  # number of rows
ncol = 5  # number of columns

discount = 0.95
eps = 1.0e-5  # should be small enough

# action: ["left", "up", "right", "down"]: [0, 1, 2, 3]

# move_increment = [np.array([0, -1]), np.array([-1, 0]), np.array([0, 1]), np.array([1, 0])]]

action_word = np.array(["left", "up", "right", "down"])

action = np.array([0, 1, 2, 3])
action_num = 4

action_prob = np.array([0.25, 0.25, 0.25, 0.25])

state_value = np.zeros(shape = (nrow, ncol))

state_action_value = np.zeros(shape = (nrow, ncol, action_num))

obstacle = [[1, 1], [1, 3], [3, 0], [3, 2]]

# greedy action expressed by one-hot encoding

greedy_action = np.zeros(shape = (nrow, ncol, action_num))

# environment

def is_terminal(state): 
	status = False
	x, y = state
	if (x == 0) and (y == 0): 
		status = True
	if (x == (nrow - 1)) and (y == (ncol - 1)):
		status = True
	return status

def get_initial():
	obstacle = [[1, 1], [1, 3], [3, 0], [3, 2]]
	bingo = True
	while bingo:
		i = np.random.randint(low = 0, high = nrow)
		j = np.random.randint(low = 0, high = ncol)
		state = [i, j]
		if is_terminal(state): 
			continue
		elif state in obstacle:
			continue
		else:
			bingo = False
	return state

from sys import exit

def step(state, action):
	obstacle = [[1, 1], [1, 3], [3, 0], [3, 2]]
	x, y = state
	if [x, y] in obstacle:
		print("The state is an obstacle!")
		print("state = ", state)
		exit()
	else:
		status = is_terminal(state)
		if status == True: 
			next_state = state
			reward = 0
		else:
			if action == 0: # move left 
				next_state = state + np.array([0, -1])
			if action == 1: # move up 
				next_state = state + np.array([-1, 0])
			if action == 2: # move right
				next_state = state + np.array([0, 1])
			if action == 3: # move down 
				next_state = state + np.array([1, 0])
			x, y = next_state
			x = np.clip(x, 0, nrow - 1)
			y = np.clip(y, 0, ncol - 1)
			if [x, y] in obstacle:
				next_state = state
			else:
				next_state[0] = x
				next_state[1] = y
			x, y = next_state
			if is_terminal([x, y]):
				reward = 0
			else:
				reward = -1
	return {"next_state": next_state, "reward": reward}

# main loop

n = 3  # n-step bootstrap

#n = 2

#n = 1  

alpha = 0.01

gamma_seg = np.ones(n+1)
for i in range(1, n+1):
	gamma_seg[i] = discount * gamma_seg[i-1]

T_big = np.inf

#episode_num = 100000

episode_num = 100000  # for testing purpose only

#episode_num = 2  # for testing purpose only

seed = 543
np.random.seed(seed)  # Set random seed for reproducibility.

for episode in range(episode_num):
	if episode == 0 or (episode+1) % 1000 == 0:
		print("episode = ", episode+1)
	state_seg = []  # segment of the state trajectory
	state_action_seg = []  # segment of the state-action trajectory
	# Get a random initial state: : Exploring starts.
	state = get_initial()
	state_seg.append(state)
	move = int(np.random.choice(action, size = 1, p = action_prob))
	state_action = [state[0], state[1], move]
	state_action_seg.append(state_action)
	T = T_big
	if is_terminal(state) == False:
		reward_seg = []  # segment of the reward trajectory
		rv_seg = np.zeros(n+1, dtype = float)
		rq_seg = np.zeros(n+1, dtype = float)
		t = 0
		go = 1
		while go == 1:
			step_obj = step(state, move)
			next_state = step_obj["next_state"]
			reward = step_obj["reward"]
			state_seg.append(next_state)
			reward_seg.append(reward)
			next_move = int(np.random.choice(action, size = 1, p = action_prob))
			next_state_action = [next_state[0], next_state[1], next_move]
			state_action_seg.append(next_state_action)
			if is_terminal(next_state):
				if T == T_big:
					T = t + 1
			#print("t = ", t, "  state = ", state, "  move = ", move, "  next_state = ", next_state, "  T = ", T)
			if t >= (n-1):
				# Update the state values.
				rv_seg[:n] = reward_seg
				k0, k1 = next_state
				rv_seg[n] = state_value[k0, k1]
				#print("rv_seg = ", rv_seg)
				GV = np.sum(gamma_seg * rv_seg)
				#print("GV = ", GV)
				k0, k1 = state_seg[0]
				#print("state value to be updated = ", state_seg[0])
				value = state_value[k0, k1] 
				value = value + alpha * (GV - value)
				state_value[k0, k1] = value
				# Update state-action values.
				rq_seg[:n] = reward_seg
				k0, k1, k2 = next_state_action
				rq_seg[n] = state_action_value[k0, k1, k2]
				#print("rq_seg = ", rq_seg)
				GQ = np.sum(gamma_seg * rq_seg)
				#print("GQ = ", GQ)
				k0, k1, k2 = state_action_seg[0]
				value = state_action_value[k0, k1, k2] 
				value = value + alpha * (GQ - value)
				state_action_value[k0, k1, k2] = value
				# Remove the first item.
				state_seg = state_seg[1:]  
				state_action_seg = state_action_seg[1:] 
				reward_seg = reward_seg[1:] 
			state = next_state
			move = next_move
			if t == (T+n-2):  
				go = 0
			else: 
				t = t + 1

# Print out the results.

np.round(state_value, 4)

np.round(state_action_value, 4)

# Find greedy actions.

for i in range(nrow):
	for j in range(ncol):
		current_state = [i, j]
		if is_terminal(current_state): 
			continue
		elif current_state in obstacle:
			continue
		else:
			max_q_value = np.max(state_action_value[i, j, :])
			index_max = which(np.abs(state_action_value[i, j, :] - max_q_value) < eps)
			greedy_action[i, j, :] = np.zeros(action_num)
			greedy_action[i, j, index_max] = 1

for i in range(nrow): 
	for j in range(ncol):
		current_state = [i, j]
		if is_terminal(current_state):
			print("state = ", [i, j], "  greedy action = ", [])
		elif current_state in obstacle:
			print("state = ", [i, j], "  greedy action = ", [])
		else:
			index_max = which(greedy_action[i, j, :] == 1)
			print("state = ", [i, j], "  greedy action = ", action_word[index_max])
