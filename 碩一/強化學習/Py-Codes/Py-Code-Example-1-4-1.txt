# Find the state values: Method 1: According to the written equations

import numpy as np

n = 5  # number of states

discount = 0.95
prob_left = 0.6
prob_right = 1 - prob_left

A = np.zeros(shape = (n, n), dtype = float)

c1 = prob_left * discount
c2 = prob_right * discount

for i in range(n):
	A[i, i] = -1

A[0, 1] = discount

for i in range(1, n-1):
	A[i, i-1] = c1
	A[i, i+1] = c2

b = np.array([1.0, 1.0, 1.0, 0.6, 0.0])

print(A)
print(b)

v = np.linalg.solve(A, b)
state_value_bellman = v

print(state_value_bellman)

# Find the state values: Method 2: More general method

# Construct state transition probability matrix and reward matrix.

import numpy as np

n = 5  # number of states

tran_prob_mat = np.zeros(shape = (n, n), dtype = float)
reward_mat = np.zeros(shape = (n, n), dtype = float)

tran_prob_mat[0, 1] = 1.0
tran_prob_mat[4, 4] = 1.0
for i in range(1, n-1):
	tran_prob_mat[i, i-1] = 0.6
	tran_prob_mat[i, i+1] = 0.4

print(tran_prob_mat)

reward_mat[:, :-1] = -1.0

print(reward_mat)

# Find state-value function by solving Bellman equation (1.4.5).

discount = 0.95

A = discount * tran_prob_mat
for i in range(n):
	A[i, i] = -1

b = (-1.0) * np.sum(tran_prob_mat * reward_mat, axis = -1)

print(A)
print(b)

v = np.linalg.solve(A, b)
state_value_bellman = v

print(state_value_bellman)
