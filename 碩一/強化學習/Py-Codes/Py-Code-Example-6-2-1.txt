# Python code for Example 6.2.1: Two-dimensional maze

# Find optimal state-action values using random-sample one-step Q-planning.

##################

# Import packages and functions.

import numpy as np

which = lambda status: np.arange(len(status))[status]

##################

nrow = 5  # number of rows
ncol = 5  # number of columns

discount = 0.95
eps = 1.0e-5  # should be small enough

#alpha = 0.01

alpha = 0.1

# action: ["left", "up", "right", "down"]: [0, 1, 2, 3]

# move_increment = [np.array([0, -1]), np.array([-1, 0]), np.array([0, 1]), np.array([1, 0])]]

action_word = np.array(["left", "up", "right", "down"])

action = np.array([0, 1, 2, 3])
action_num = 4

#action_prob = np.array([0.25, 0.25, 0.25, 0.25])

state_action_value = np.zeros(shape = (nrow, ncol, action_num))

obstacle = [[1, 1], [1, 3], [3, 0], [3, 2]]

# greedy action expressed by one-hot encoding

greedy_action = np.zeros(shape = (nrow, ncol, action_num))

# environment

def is_terminal(state): 
	status = False
	x, y = state
	if (x == 0) and (y == 0): 
		status = True
	if (x == (nrow - 1)) and (y == (ncol - 1)):
		status = True
	return status

def get_initial():
	obstacle = [[1, 1], [1, 3], [3, 0], [3, 2]]
	bingo = True
	while bingo:
		i = np.random.randint(low = 0, high = nrow)
		j = np.random.randint(low = 0, high = ncol)
		state = [i, j]
		if is_terminal(state): 
			continue
		elif state in obstacle:
			continue
		else:
			bingo = False
	return state

from sys import exit

def step(state, action):
	obstacle = [[1, 1], [1, 3], [3, 0], [3, 2]]
	x, y = state
	if [x, y] in obstacle:
		print("The state is an obstacle!")
		print("state = ", state)
		exit()
	else:
		status = is_terminal(state)
		if status == True: 
			next_state = state
			reward = 0
		else:
			if action == 0: # move left 
				next_state = state + np.array([0, -1])
			if action == 1: # move up 
				next_state = state + np.array([-1, 0])
			if action == 2: # move right
				next_state = state + np.array([0, 1])
			if action == 3: # move down 
				next_state = state + np.array([1, 0])
			x, y = next_state
			x = np.clip(x, 0, nrow - 1)
			y = np.clip(y, 0, ncol - 1)
			if [x, y] in obstacle:
				next_state = state
			else:
				next_state[0] = x
				next_state[1] = y
			x, y = next_state
			if is_terminal([x, y]):
				reward = 0
			else:
				reward = -1
	return {"next_state": next_state, "reward": reward}

# main loop

#iteration_num = 100000

iteration_num = 10000  # for testing purpose only

num_visits = np.zeros(shape = (nrow, ncol, action_num), dtype = int)

seed = 543
np.random.seed(seed)  # Set random seed for reproducibility.

for iteration in range(iteration_num):
	if iteration == 0 or (iteration+1) % 1000 == 0:
		print("iteration = ", iteration+1)
	# Randomly choose a state: Exploring starts.
	state = get_initial()
	# Randomly choose an action 
	#move = int(np.random.choice(action, size = 1, p = action_prob))
	move = int(np.random.choice(action, size = 1))
	num_visits[state[0], state[1], move] = num_visits[state[0], state[1], move] + 1
	# Update the state-action value.
	step_obj = step(state, move)
	next_state = list(step_obj["next_state"])
	reward = step_obj["reward"]
	value = state_action_value[state[0], state[1], move]
	next_opt_value = np.max(state_action_value[next_state[0], next_state[1], :])
	value = value + alpha * (reward + discount * next_opt_value - value)
	state_action_value[state[0], state[1], move] = value

# Print out the results.

np.round(state_action_value, 4)

num_visits

# Find greedy actions.

for i in range(nrow):
	for j in range(ncol):
		current_state = [i, j]
		if is_terminal(current_state): 
			continue
		elif current_state in obstacle:
				continue
		else:
			max_q_value = np.max(state_action_value[i, j, :])
			index_max = which(np.abs(state_action_value[i, j, :] - max_q_value) < eps)
			greedy_action[i, j, :] = np.zeros(action_num)
			greedy_action[i, j, index_max] = 1

for i in range(nrow): 
	for j in range(ncol):
		current_state = [i, j]
		if is_terminal(current_state):
			print("state = ", [i, j], "  greedy action = ", [])
		else:
			index_max = which(greedy_action[i, j, :] == 1)
			print("state = ", [i, j], "  greedy action = ", action_word[index_max])
