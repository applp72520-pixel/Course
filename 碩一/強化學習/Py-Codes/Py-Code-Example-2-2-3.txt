# Python code for Example 2.2.3: One-dimensional Markov decision process

# Find optimal state values and optimal state-action values using policy iteration.

##################

# Import packages and functions.

import numpy as np

which = lambda status: np.arange(len(status))[status]

##################

n = 5  # number of states

discount = 0.95
eps = 1.0e-5

# action: ["left", "right"]: [0, 1]

action_word = np.array(["left", "right"])

action = np.array([0, 1])
action_num = 2

action_prob = np.zeros(shape = (n, action_num))
action_prob[:, :] = 0.5
action_prob[n-1, :] = 0.0

state_value = np.zeros(n)
state_value_old = np.zeros(n)

state_action_value = np.zeros(shape = (n, action_num))
state_action_value_old = np.zeros(shape = (n, action_num))

# greedy action expressed by one-hot encoding

greedy_action = np.zeros(shape = (n, action_num))

# environment

def is_terminal(state): 
	x = state
	status = False
	if x == 4:
		status = True
	return status

def step(state, action):
	x = state
	if is_terminal(x):
		next_state = x
		reward = 0.0
	elif x == 0:
		next_state = x + 1
		reward = -1.0
	else:
		reward = -1.0
		if action == 0: # move left 
			next_state = x - 1
		if action == 1: # move right 
			next_state = x + 1
		if is_terminal(next_state):
			reward = 0.0
	return {"next_state": next_state, "reward": reward}

# main loop

go = 1; iteration = 0

while go == 1:
	iteration = iteration + 1
	#print("iteration = ", iteration, "\n")
	# policy evaluation
	for i in range(n):
		current_state = i
		if is_terminal(current_state):
			continue
		else:
			for k in range(action_num):
				move = k
				step_obj = step(current_state, move)
				next_state = step_obj["next_state"]
				reward = step_obj["reward"]
				state_action_value[i, k] = reward + discount * state_value[next_state] #求出q值
			state_value[i] = np.sum(action_prob[i, :] * state_action_value[i, :])   #求出v值
			# Find greedy actions.
			max_q_value = np.max(state_action_value[i])
			index_max = which(np.abs(state_action_value[i] - max_q_value) < eps)
			greedy_action[i, :] = np.zeros(action_num)
			greedy_action[i, index_max] = 1
			# Print out the results.
			#print("current state = ", current_state, "  state value = ", np.round(state_value[i]), "  greedy action = ", greedy_action[i, :])
	#print(np.round(state_value, 4), "\n")
	max_abs_diff = np.max(np.abs(state_value - state_value_old))
	if max_abs_diff < eps: 
		go = 0
	else:
		state_value_old = np.copy(state_value)
		# policy improvement
		for i in range(n):
			current_state = i
			if is_terminal(current_state):
				continue
			else:
				index_max = which(greedy_action[i, :] == 1)
				index_max_length = len(index_max)
				action_prob[i, :] = np.zeros(action_num) #改變策略
				action_prob[i, index_max] = 1.0 / index_max_length

iteration

print(np.round(state_value, 4))

print(np.round(state_action_value, 4))

print(np.round(action_prob, 4))

for i in range(n): 
	if is_terminal(i):
		print("state = ", i, "  greedy action = ", [])
	else:
		index_max = which(greedy_action[i, :] == 1)
		print("state = ", i, "  greedy action = ", action_word[index_max])
